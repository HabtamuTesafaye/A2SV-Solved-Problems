class Solution:
    def dividePlayers(self, skill: List[int]) -> int:
        skill.sort()
        n = len(skill)

        target_sum = skill[0] + skill[-1]
        chemistry_sum = 0

        left,right = 0, n -1
        while left < right:
            current_sum  = skill[left] + skill[right]

            # Check if this pair has the target sum
            if current_sum != target_sum:
                return -1

            chemistry_sum += skill[left] * skill[right]

            left += 1
            right -= 1

        return chemistry_sum
